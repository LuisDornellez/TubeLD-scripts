        var r = (cnt ? sum / cnt / 255 : 0) * GAIN[i];
        raw.push(r);
        if (r > mx) mx = r;
      }
      vpk = Math.max(mx, vpk * 0.995, 0.25);
      var o = [];
      for (i = 0; i < raw.length; i++) { o.push(Math.min(1, Math.pow(raw[i] / vpk, 1.15)).toFixed(2)); }
      window.TLD.lv(o.join(','));
    } catch (e) { L('eq.viz', e); }
  }, 80);
